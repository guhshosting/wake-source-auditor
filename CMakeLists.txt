cmake_minimum_required(VERSION 3.24)
project(wake-source-auditor VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC are required")
endif()
option(WSA_ANALYZE "Enable MSVC static analysis" OFF)
option(WSA_ASAN "Enable MSVC AddressSanitizer" OFF)
if(WSA_ANALYZE AND WSA_ASAN)
  message(FATAL_ERROR "Use separate analysis and ASan builds")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Standalone release executable: no separate Visual C++ runtime installation.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(wsa_core STATIC src/model.cpp src/explain.cpp src/event_parser.cpp src/format.cpp src/power.cpp src/events.cpp src/devices.cpp src/tasks.cpp)
target_include_directories(wsa_core PUBLIC include)
target_compile_definitions(wsa_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(wsa_core PUBLIC powrprof wevtapi setupapi ole32 oleaut32 taskschd xmllite shlwapi)
add_executable(wake-source-auditor src/main.cpp src/version.rc)
target_link_libraries(wake-source-auditor PRIVATE wsa_core)
target_link_options(wake-source-auditor PRIVATE /MANIFESTUAC:level='asInvoker' /DYNAMICBASE /NXCOMPAT)
include(CTest)
if(BUILD_TESTING)
  add_executable(wsa_tests tests/tests.cpp)
  target_link_libraries(wsa_tests PRIVATE wsa_core)
  add_test(NAME synthetic COMMAND wsa_tests)
  add_test(NAME integration COMMAND wsa_tests --integration)
  set_tests_properties(integration PROPERTIES LABELS integration TIMEOUT 90)
  add_test(NAME cli_help COMMAND wake-source-auditor --help)
  add_test(NAME cli_version COMMAND wake-source-auditor --version)
  add_test(NAME cli_json COMMAND wake-source-auditor --json)
  set_tests_properties(cli_json PROPERTIES TIMEOUT 90)
endif()
foreach(t wsa_core wake-source-auditor wsa_tests)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
    if(WSA_ANALYZE)
      # SDK 26100 powrprof.h/XmlLite use pointer SAL on scalar parameters.
      # C6553 diagnoses those declarations, not application data flow.
      target_compile_options(${t} PRIVATE /analyze /wd6553)
    endif()
    if(WSA_ASAN)
      target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
      target_link_options(${t} PRIVATE /INCREMENTAL:NO)
    endif()
  endif()
endforeach()
install(TARGETS wake-source-auditor RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/wake-source-auditor)
install(FILES docs/JSON.md docs/API-SOURCES.md DESTINATION share/wake-source-auditor/docs)
install(FILES validation/REPORT.md DESTINATION share/wake-source-auditor/validation)
